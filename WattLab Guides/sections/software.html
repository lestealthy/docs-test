Task Scheduling

RTOS scheduling algorithms determine which task runs when:

Preemptive Priority Scheduling 
void rtos_scheduler(void) {
    rtos_task_t *highest_priority = NULL;
    rtos_task_t *current_task = get_current_task();
    
    // Find highest priority ready task
    for (int i = 0; i < MAX_TASKS; i++) {
        if (task_list[i].state == TASK_READY) {
            if (highest_priority == NULL || 
                task_list[i].priority > highest_priority->priority) {
                highest_priority = &task_list[i];
            }
        }
    }
    
    // Context switch if needed
    if (highest_priority != current_task) {
        context_switch(current_task, highest_priority);
    }
}

typedef struct {
    uint8_t *pool;
    uint32_t size;
    uint32_t used;
    uint32_t free_blocks;
} memory_pool_t;

void* pool_alloc(memory_pool_t *pool, size_t size) {
    if (pool->used + size > pool->size) {
        return NULL; // Out of memory
    }
    
    void *ptr = &pool->pool[pool->used];
    pool->used += size;
    return ptr;
}

void nvic_enable_irq(uint8_t irq_num) {
    volatile uint32_t *iser = (volatile uint32_t*)NVIC_ISER;
    SET_BIT(*iser, irq_num);
}

void nvic_set_priority(uint8_t irq_num, uint8_t priority) {
    volatile uint32_t *ipr = (volatile uint32_t*)(NVIC_IPR + (irq_num / 4));
    uint8_t shift = (irq_num % 4) * 8;
    *ipr = (*ipr & ~(0xFF << shift)) | (priority << shift);
}

Embedded Design Patterns

Observer Pattern

For event-driven systems:

Observer Pattern Implementation 
typedef struct observer observer_t;
typedef struct subject subject_t;

typedef void (*observer_callback_t)(observer_t *obs, void *data);

struct observer {
    observer_callback_t callback;
    observer_t *next;
};

struct subject {
    observer_t *observers;
    void *data;
};

void subject_attach(subject_t *subject, observer_t *observer) {
    observer->next = subject->observers;
    subject->observers = observer;
}

void subject_notify(subject_t *subject) {
    observer_t *obs = subject->observers;
    while (obs) {
        obs->callback(obs, subject->data);
        obs = obs->next;
    }
}

State Machine Pattern

For complex system behavior:

State Machine Implementation 
typedef enum {
    STATE_IDLE,
    STATE_ACTIVE,
    STATE_ERROR
} system_state_t;

typedef struct {
    system_state_t current_state;
    system_state_t next_state;
    uint32_t state_timer;
} state_machine_t;

typedef void (*state_handler_t)(state_machine_t *sm);

static void idle_handler(state_machine_t *sm) {
    if (start_condition()) {
        sm->next_state = STATE_ACTIVE;
    }
}

static void active_handler(state_machine_t *sm) {
    if (error_condition()) {
        sm->next_state = STATE_ERROR;
    } else if (stop_condition()) {
        sm->next_state = STATE_IDLE;
    }
}

static void error_handler(state_machine_t *sm) {
    if (reset_condition()) {
        sm->next_state = STATE_IDLE;
    }
}

static state_handler_t state_handlers[] = {
    idle_handler,
    active_handler,
    error_handler
};

void state_machine_update(state_machine_t *sm) {
    state_handlers[sm->current_state](sm);
    sm->current_state = sm->next_state;
}

    int len = sprintf(value_str, "%d\n", gpio_value);

static struct file_operations fops = {
    .owner = THIS_MODULE,
    .read = device_read,
};

static struct miscdevice custom_misc_device = {
    .minor = MISC_DYNAMIC_MINOR,
    .name = DEVICE_NAME,
    .fops = &fops,
};

static int __init custom_init(void) {
    int ret;
    
    if (gpio_request(GPIO_PIN, "custom_gpio")) {
        printk(KERN_ERR "Failed to request GPIO %d\n", GPIO_PIN);
        return -1;
    }
    
    gpio_direction_input(GPIO_PIN);
    gpio_irq = gpio_to_irq(GPIO_PIN);
    
    ret = request_irq(gpio_irq, gpio_interrupt_handler, 
                      IRQF_TRIGGER_RISING | IRQF_TRIGGER_FALLING,
                      "custom_gpio", NULL);
    if (ret) {
        printk(KERN_ERR "Failed to request IRQ %d\n", gpio_irq);
        gpio_free(GPIO_PIN);
        return ret;
    }
    
    ret = misc_register(&custom_misc_device);
    if (ret) {
        printk(KERN_ERR "Failed to register misc device\n");
        free_irq(gpio_irq, NULL);
        gpio_free(GPIO_PIN);
        return ret;
    }
    
    printk(KERN_INFO "Custom device driver loaded\n");
    return 0;
}

static void __exit custom_exit(void) {
    misc_deregister(&custom_misc_device);
    free_irq(gpio_irq, NULL);
    gpio_free(GPIO_PIN);
    printk(KERN_INFO "Custom device driver unloaded\n");
}

module_exit(custom_exit);

Memory Optimization

Techniques for memory-constrained systems:

Memory Pool Implementation 
typedef struct {
    uint8_t *memory;
    size_t block_size;
    size_t total_blocks;
    uint32_t *free_blocks;
    uint32_t free_count;
} memory_pool_t;

void pool_init(memory_pool_t *pool, uint8_t *memory, 
               size_t block_size, size_t total_blocks) {
    pool->memory = memory;
    pool->block_size = block_size;
    pool->total_blocks = total_blocks;
    pool->free_count = total_blocks;
    
    // Initialize free block list
    for (size_t i = 0; i < total_blocks; i++) {
        pool->free_blocks[i] = i;
    }
}

void* pool_alloc(memory_pool_t *pool) {
    if (pool->free_count == 0) {
        return NULL; // Out of memory
    }
    
    uint32_t block_index = pool->free_blocks[--pool->free_count];
    return &pool->memory[block_index * pool->block_size];
}

void pool_free(memory_pool_t *pool, void *ptr) {
    if (ptr == NULL) return;
    
    uint32_t block_index = ((uint8_t*)ptr - pool->memory) / pool->block_size;
    pool->free_blocks[pool->free_count++] = block_index;
}

Real-Time Constraints

Meeting timing requirements:

Real-Time Task Implementation 
typedef struct {
    uint32_t deadline;
    uint32_t period;
    uint32_t execution_time;
    uint32_t priority;
    void (*function)(void);
} realtime_task_t;

// Rate Monotonic Scheduling
void rate_monotonic_schedule(realtime_task_t *tasks, int num_tasks) {
    // Sort by period (shortest period = highest priority)
    for (int i = 0; i < num_tasks - 1; i++) {
        for (int j = 0; j < num_tasks - i - 1; j++) {
            if (tasks[j].period > tasks[j + 1].period) {
                realtime_task_t temp = tasks[j];
                tasks[j] = tasks[j + 1];
                tasks[j + 1] = temp;
            }
        }
    }
}

// Deadline Monotonic Scheduling
bool deadline_monotonic_feasible(realtime_task_t *tasks, int num_tasks) {
    for (int i = 0; i < num_tasks; i++) {
        uint32_t utilization = 0;
        for (int j = 0; j <= i; j++) {
            utilization += (tasks[j].execution_time * 100) / tasks[j].deadline;
        }
        if (utilization > 100) {
            return false; // Not schedulable
        }
    }
    return true;
}
