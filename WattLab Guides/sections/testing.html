// Test framework for embedded systems
typedef struct {
    const char *name;
    void (*test_func)(void);
    bool passed;
    uint32_t execution_time;
} test_case_t;

typedef struct {
    test_case_t *tests;
    uint32_t test_count;
    uint32_t passed_count;
    uint32_t failed_count;
    uint32_t total_time;
} test_suite_t;

static test_suite_t g_test_suite = {0};

// Test assertions
#define ASSERT_TRUE(condition) \
    do { \
        if (!(condition)) { \
            printk("ASSERTION FAILED: %s at %s:%d\n", #condition, __FILE__, __LINE__); \
            return false; \
        } \
    } while(0)

#define ASSERT_FALSE(condition) \
    do { \
        if (condition) { \
            printk("ASSERTION FAILED: %s at %s:%d\n", #condition, __FILE__, __LINE__); \
            return false; \
        } \
    } while(0)

#define ASSERT_EQUAL(expected, actual) \
    do { \
        if ((expected) != (actual)) { \
            printk("ASSERTION FAILED: Expected %d, got %d at %s:%d\n", \
                   (int)(expected), (int)(actual), __FILE__, __LINE__); \
            return false; \
        } \
    } while(0)

#define ASSERT_FLOAT_EQUAL(expected, actual, tolerance) \
    do { \
        float diff = (expected) - (actual); \
        if (diff < 0) diff = -diff; \
        if (diff > (tolerance)) { \
            printk("ASSERTION FAILED: Expected %f, got %f at %s:%d\n", \
                   (expected), (actual), __FILE__, __LINE__); \
            return false; \
        } \
    } while(0)

// Test registration
void register_test(const char *name, void (*test_func)(void)) {
    if (g_test_suite.test_count < MAX_TESTS) {
        g_test_suite.tests[g_test_suite.test_count].name = name;
        g_test_suite.tests[g_test_suite.test_count].test_func = test_func;
        g_test_suite.test_count++;
    }
}

// Test execution
bool run_test(test_case_t *test) {
    uint32_t start_time = k_uptime_get();
    
    printk("Running test: %s\n", test->name);
    
    // Set up test environment
    test->passed = true;
    
    // Run test function
    test->test_func();
    
    test->execution_time = k_uptime_get() - start_time;
    
    if (test->passed) {
        printk("PASS: %s (%u ms)\n", test->name, test->execution_time);
        g_test_suite.passed_count++;
    } else {
        printk("FAIL: %s (%u ms)\n", test->name, test->execution_time);
        g_test_suite.failed_count++;
    }
    
    return test->passed;
}

// Test suite execution
void run_all_tests(void) {
    printk("Starting test suite...\n");
    g_test_suite.passed_count = 0;
    g_test_suite.failed_count = 0;
    g_test_suite.total_time = 0;
    
    uint32_t start_time = k_uptime_get();
    
    for (uint32_t i = 0; i < g_test_suite.test_count; i++) {
        run_test(&g_test_suite.tests[i]);
        g_test_suite.total_time += g_test_suite.tests[i].execution_time;
    }
    
    uint32_t total_time = k_uptime_get() - start_time;
    
    printk("\nTest Results:\n");
    printk("Total tests: %u\n", g_test_suite.test_count);
    printk("Passed: %u\n", g_test_suite.passed_count);
    printk("Failed: %u\n", g_test_suite.failed_count);
    printk("Total time: %u ms\n", total_time);
    printk("Success rate: %.1f%%\n", 
           (float)g_test_suite.passed_count / g_test_suite.test_count * 100.0f);
}

void mock_uart_send(const char *data, size_t len) {
    if (len > sizeof(g_mock_uart.buffer) - g_mock_uart.write_pos) {
        g_mock_uart.overflow = true;
        return;
    }
    
    memcpy(&g_mock_uart.buffer[g_mock_uart.write_pos], data, len);
    g_mock_uart.write_pos += len;
}

size_t mock_uart_receive(char *data, size_t max_len) {
    size_t available = g_mock_uart.write_pos - g_mock_uart.read_pos;
    size_t to_read = (available < max_len) ? available : max_len;
    
    memcpy(data, &g_mock_uart.buffer[g_mock_uart.read_pos], to_read);
    g_mock_uart.read_pos += to_read;
    
    return to_read;
}

static hil_test_harness_t g_test_harness = {0};

// Initialize HIL test harness
int hil_test_init(void) {
    g_test_harness.gpio_dev = device_get_binding("GPIO_0");
    g_test_harness.uart_dev = device_get_binding("UART_0");
    g_test_harness.i2c_dev = device_get_binding("I2C_0");
    g_test_harness.spi_dev = device_get_binding("SPI_0");
    
    if (!g_test_harness.gpio_dev || !g_test_harness.uart_dev) {
        return -ENODEV;
    }
    
    return 0;
}

// Power supply testing
hil_test_results_t test_power_supply_regulation(hil_test_conditions_t *conditions) {
    hil_test_results_t results = {0};
    
    // Set input voltage
    set_power_supply_voltage(conditions->input_voltage);
    k_sleep(K_MSEC(100));
    
    // Measure output voltage
    float measured_voltage = measure_output_voltage();
    results.measured_values[0] = measured_voltage;
    
    // Check regulation accuracy
    float voltage_error = fabs(measured_voltage - conditions->output_voltage);
    if (voltage_error > 0.1f) {
        results.test_passed = false;
        results.error_count++;
        snprintf(results.error_messages[results.error_count-1], 64,
                "Voltage regulation error: %.3fV", voltage_error);
    }
    
    // Test load regulation
    set_load_current(conditions->current);
    k_sleep(K_MSEC(50));
    
    float loaded_voltage = measure_output_voltage();
    results.measured_values[1] = loaded_voltage;
    
    float load_regulation = fabs(loaded_voltage - measured_voltage);
    if (load_regulation > 0.05f) {
        results.test_passed = false;
        results.error_count++;
        snprintf(results.error_messages[results.error_count-1], 64,
                "Load regulation error: %.3fV", load_regulation);
    }
    
    return results;
}

// Communication interface testing
hil_test_results_t test_uart_communication(void) {
    hil_test_results_t results = {0};
    
    const char test_patterns[][16] = {
        "Hello World",
        "0123456789",
        "ABCDEFGHIJ",
        "!@#$%^&*()",
        ""
    };
    
    for (int i = 0; i < 4; i++) {
        // Send test pattern
        uart_poll_out(g_test_harness.uart_dev, test_patterns[i], strlen(test_patterns[i]));
        k_sleep(K_MSEC(10));
        
        // Receive and verify
        char received[16];
        int received_len = 0;
        char c;
        
        while (uart_poll_in(g_test_harness.uart_dev, &c) == 0 && received_len < 15) {
            received[received_len++] = c;
        }
        received[received_len] = '\0';
        
        if (strcmp(test_patterns[i], received) != 0) {
            results.test_passed = false;
            results.error_count++;
            snprintf(results.error_messages[results.error_count-1], 64,
                    "UART pattern %d mismatch", i);
        }
    }
    
    return results;
}

// I2C device testing
hil_test_results_t test_i2c_devices(void) {
    hil_test_results_t results = {0};
    
    // Test I2C device discovery
    uint8_t device_addresses[10];
    int device_count = i2c_scan_devices(g_test_harness.i2c_dev, device_addresses, 10);
    
    if (device_count == 0) {
        results.test_passed = false;
        results.error_count++;
        strcpy(results.error_messages[results.error_count-1], "No I2C devices found");
        return results;
    }
    
    // Test communication with each device
    for (int i = 0; i < device_count; i++) {
        uint8_t addr = device_addresses[i];
        
        // Write test data
        uint8_t write_data[] = {0x00, 0x55, 0xAA};
        int ret = i2c_write(g_test_harness.i2c_dev, write_data, sizeof(write_data), addr);
        
        if (ret != 0) {
            results.test_passed = false;
            results.error_count++;
            snprintf(results.error_messages[results.error_count-1], 64,
                    "I2C write failed for device 0x%02X", addr);
        }
        
        k_sleep(K_MSEC(10));
        
        // Read back data
        uint8_t read_data[3];
        ret = i2c_read(g_test_harness.i2c_dev, read_data, sizeof(read_data), addr);
        
        if (ret != 0) {
            results.test_passed = false;
            results.error_count++;
            snprintf(results.error_messages[results.error_count-1], 64,
                    "I2C read failed for device 0x%02X", addr);
        }
    }
    
    return results;
}

// GPIO testing
hil_test_results_t test_gpio_functionality(void) {
    hil_test_results_t results = {0};
    
    // Test GPIO output
    for (int pin = 0; pin < 8; pin++) {
        gpio_pin_configure(g_test_harness.gpio_dev, pin, GPIO_OUTPUT);
        
        // Test high output
        gpio_pin_set(g_test_harness.gpio_dev, pin, 1);
        k_sleep(K_MSEC(1));
        
        int level = gpio_pin_get(g_test_harness.gpio_dev, pin);
        if (level != 1) {
            results.test_passed = false;
            results.error_count++;
            snprintf(results.error_messages[results.error_count-1], 64,
                    "GPIO pin %d high test failed", pin);
        }
        
        // Test low output
        gpio_pin_set(g_test_harness.gpio_dev, pin, 0);
        k_sleep(K_MSEC(1));
        
        level = gpio_pin_get(g_test_harness.gpio_dev, pin);
        if (level != 0) {
            results.test_passed = false;
            results.error_count++;
            snprintf(results.error_messages[results.error_count-1], 64,
                    "GPIO pin %d low test failed", pin);
        }
    }
    
    return results;
}

typedef struct {
    uint32_t start_time;
    uint32_t end_time;
    uint32_t execution_time;
    uint32_t max_time;
    uint32_t min_time;
    uint32_t call_count;
    float average_time;
    const char *function_name;
} performance_counter_t;

#define MAX_PERFORMANCE_COUNTERS 20
static performance_counter_t g_perf_counters[MAX_PERFORMANCE_COUNTERS];
static uint32_t g_perf_counter_count = 0;

// Performance measurement macros
#define PERF_START(name) \
    static performance_counter_t *perf_counter = get_performance_counter(name); \
    uint32_t perf_start_time = k_uptime_get()

#define PERF_END() \
    do { \
        uint32_t perf_end_time = k_uptime_get(); \
        update_performance_counter(perf_counter, perf_start_time, perf_end_time); \
    } while(0)

// Get or create performance counter
static performance_counter_t *get_performance_counter(const char *name) {
    // Find existing counter
    for (uint32_t i = 0; i < g_perf_counter_count; i++) {
        if (strcmp(g_perf_counters[i].function_name, name) == 0) {
            return &g_perf_counters[i];
        }
    }
    
    // Create new counter
    if (g_perf_counter_count < MAX_PERFORMANCE_COUNTERS) {
        performance_counter_t *counter = &g_perf_counters[g_perf_counter_count++];
        counter->function_name = name;
        counter->call_count = 0;
        counter->max_time = 0;
        counter->min_time = UINT32_MAX;
        counter->average_time = 0.0f;
        return counter;
    }
    
    return NULL;
}

// Update performance counter
static void update_performance_counter(performance_counter_t *counter, 
                                     uint32_t start_time, uint32_t end_time) {
    if (!counter) return;
    
    uint32_t execution_time = end_time - start_time;
    
    counter->call_count++;
    counter->execution_time = execution_time;
    
    if (execution_time > counter->max_time) {
        counter->max_time = execution_time;
    }
    
    if (execution_time < counter->min_time) {
        counter->min_time = execution_time;
    }
    
    // Update running average
    counter->average_time = (counter->average_time * (counter->call_count - 1) + 
                           execution_time) / counter->call_count;
}

// Memory usage tracking
typedef struct {
    uint32_t heap_used;
    uint32_t heap_total;
    uint32_t stack_used;
    uint32_t stack_total;
    uint32_t static_used;
    uint32_t static_total;
} memory_stats_t;

static memory_stats_t get_memory_stats(void) {
    memory_stats_t stats = {0};
    
    // Get heap statistics
    struct k_mem_slab *heap_slab;
    k_mem_slab_get_info(&heap_slab, &stats.heap_used, &stats.heap_total);
    
    // Get stack usage (approximate)
    stats.stack_used = CONFIG_MAIN_STACK_SIZE - 
                      (uint32_t)__builtin_frame_address(0) % CONFIG_MAIN_STACK_SIZE;
    stats.stack_total = CONFIG_MAIN_STACK_SIZE;
    
    // Get static memory usage
    extern char _bss_start[], _bss_end[];
    extern char _data_start[], _data_end[];
    stats.static_used = (uint32_t)(_bss_end - _bss_start) + 
                       (uint32_t)(_data_end - _data_start);
    stats.static_total = CONFIG_SRAM_SIZE;
    
    return stats;
}

// CPU usage monitoring
typedef struct {
    uint32_t idle_time;
    uint32_t total_time;
    uint32_t last_idle_time;
    uint32_t last_total_time;
    float cpu_usage;
} cpu_monitor_t;

static cpu_monitor_t g_cpu_monitor = {0};

static void update_cpu_usage(void) {
    uint32_t current_idle_time = k_uptime_get();
    uint32_t current_total_time = k_uptime_get();
    
    uint32_t idle_delta = current_idle_time - g_cpu_monitor.last_idle_time;
    uint32_t total_delta = current_total_time - g_cpu_monitor.last_total_time;
    
    if (total_delta > 0) {
        g_cpu_monitor.cpu_usage = 100.0f * (1.0f - (float)idle_delta / total_delta);
    }
    
    g_cpu_monitor.last_idle_time = current_idle_time;
    g_cpu_monitor.last_total_time = current_total_time;
}

// Real-time performance analysis
typedef struct {
    uint32_t deadline_misses;
    uint32_t total_deadlines;
    float deadline_miss_rate;
    uint32_t max_response_time;
    uint32_t min_response_time;
    float average_response_time;
} realtime_stats_t;

static realtime_stats_t analyze_realtime_performance(void) {
    realtime_stats_t stats = {0};
    
    // Analyze task execution times
    for (uint32_t i = 0; i < g_perf_counter_count; i++) {
        performance_counter_t *counter = &g_perf_counters[i];
        
        // Check for deadline misses (assuming 10ms deadline)
        uint32_t deadline = 10; // 10ms
        if (counter->max_time > deadline) {
            stats.deadline_misses++;
        }
        stats.total_deadlines++;
        
        // Update response time statistics
        if (counter->max_time > stats.max_response_time) {
            stats.max_response_time = counter->max_time;
        }
        
        if (counter->min_time < stats.min_response_time) {
            stats.min_response_time = counter->min_time;
        }
        
        stats.average_response_time += counter->average_time;
    }
    
    if (stats.total_deadlines > 0) {
        stats.deadline_miss_rate = (float)stats.deadline_misses / stats.total_deadlines * 100.0f;
        stats.average_response_time /= g_perf_counter_count;
    }
    
    return stats;
}

// Performance report generation
void generate_performance_report(void) {
    printk("\n=== Performance Report ===\n");
    
    // Function performance
    printk("Function Performance:\n");
    for (uint32_t i = 0; i < g_perf_counter_count; i++) {
        performance_counter_t *counter = &g_perf_counters[i];
        printk("  %s: %u calls, avg: %.2f ms, max: %u ms, min: %u ms\n",
               counter->function_name, counter->call_count, 
               counter->average_time, counter->max_time, counter->min_time);
    }
    
    // Memory usage
    memory_stats_t mem_stats = get_memory_stats();
    printk("\nMemory Usage:\n");
    printk("  Heap: %u/%u bytes (%.1f%%)\n", 
           mem_stats.heap_used, mem_stats.heap_total,
           (float)mem_stats.heap_used / mem_stats.heap_total * 100.0f);
    printk("  Stack: %u/%u bytes (%.1f%%)\n",
           mem_stats.stack_used, mem_stats.stack_total,
           (float)mem_stats.stack_used / mem_stats.stack_total * 100.0f);
    printk("  Static: %u/%u bytes (%.1f%%)\n",
           mem_stats.static_used, mem_stats.static_total,

// Trace entry
typedef struct {

// Debug macros
#define DEBUG_LOG(level, fmt, ...) \
    do { \
        if (level <= g_debug_context.level) { \
            LOG_##level(fmt, ##__VA_ARGS__); \
            if (g_debug_context.enable_tracing) { \

            } \
        } \
    } while(0)

// Add trace entry
static void add_trace_entry(debug_level_t level, const char *function, 
                           const char *file, uint32_t line, const char *fmt, ...) {
    if (g_trace_index >= g_debug_context.trace_buffer_size) {
        g_trace_index = 0;  // Wrap around
    }
    
    trace_entry_t *entry = &g_trace_buffer[g_trace_index++];
    entry->timestamp = k_uptime_get();
    entry->function = function;
    entry->file = file;
    entry->line = line;
    entry->level = level;
    
    va_list args;
    va_start(args, fmt);
    vsnprintf(entry->message, sizeof(entry->message), fmt, args);
    va_end(args);
    
    g_debug_context.trace_count++;
}

// Memory debugging
typedef struct {
    void *ptr;
    size_t size;
    const char *file;
    uint32_t line;
    uint32_t timestamp;
} memory_allocation_t;

#define MAX_MEMORY_TRACKING 100
static memory_allocation_t g_memory_tracking[MAX_MEMORY_TRACKING];
static uint32_t g_memory_count = 0;

// Tracked memory allocation
void *debug_malloc(size_t size, const char *file, uint32_t line) {
    void *ptr = k_malloc(size);
    
    if (ptr && g_debug_context.enable_memory_tracking) {
        if (g_memory_count < MAX_MEMORY_TRACKING) {
            g_memory_tracking[g_memory_count].ptr = ptr;
            g_memory_tracking[g_memory_count].size = size;
            g_memory_tracking[g_memory_count].file = file;
            g_memory_tracking[g_memory_count].line = line;
            g_memory_tracking[g_memory_count].timestamp = k_uptime_get();
            g_memory_count++;
        }
        
        DEBUG_TRACE("Memory allocated: %p, size: %u at %s:%u", ptr, size, file, line);
    }
    
    return ptr;
}

// Tracked memory deallocation
void debug_free(void *ptr, const char *file, uint32_t line) {
    if (g_debug_context.enable_memory_tracking) {
        // Find and remove from tracking
        for (uint32_t i = 0; i < g_memory_count; i++) {
            if (g_memory_tracking[i].ptr == ptr) {
                DEBUG_TRACE("Memory freed: %p at %s:%u", ptr, file, line);
                
                // Remove from tracking
                for (uint32_t j = i; j < g_memory_count - 1; j++) {
                    g_memory_tracking[j] = g_memory_tracking[j + 1];
                }
                g_memory_count--;
                break;
            }
        }
    }
    
    k_free(ptr);
}

// Memory leak detection
void check_memory_leaks(void) {
    if (!g_debug_context.enable_memory_tracking) return;
    
    DEBUG_INFO("Checking for memory leaks...");
    
    uint32_t total_allocated = 0;
    for (uint32_t i = 0; i < g_memory_count; i++) {
        total_allocated += g_memory_tracking[i].size;
        DEBUG_WARNING("Memory leak: %p, size: %u, allocated at %s:%u",
                     g_memory_tracking[i].ptr, g_memory_tracking[i].size,
                     g_memory_tracking[i].file, g_memory_tracking[i].line);
    }
    
    if (g_memory_count > 0) {
        DEBUG_ERROR("Memory leak detected: %u allocations, %u bytes", 
                   g_memory_count, total_allocated);
    } else {
        DEBUG_INFO("No memory leaks detected");
    }
}

// Stack overflow detection
void check_stack_overflow(void) {
    extern char _stack_start[], _stack_end[];
    char *stack_ptr = (char*)__builtin_frame_address(0);
    
    size_t stack_used = (size_t)(_stack_end - stack_ptr);
    size_t stack_total = (size_t)(_stack_end - _stack_start);
    float stack_usage = (float)stack_used / stack_total * 100.0f;
    
    DEBUG_INFO("Stack usage: %u/%u bytes (%.1f%%)", 
               stack_used, stack_total, stack_usage);
    
    if (stack_usage > 80.0f) {
        DEBUG_WARNING("High stack usage: %.1f%%", stack_usage);
    }
    
    if (stack_usage > 95.0f) {
        DEBUG_ERROR("Stack overflow risk: %.1f%%", stack_usage);
    }
}

// Debug trace dump
void dump_debug_trace(void) {
    DEBUG_INFO("=== Debug Trace Dump ===");
    DEBUG_INFO("Total trace entries: %u", g_debug_context.trace_count);
    
    uint32_t start_index = (g_trace_index >= g_debug_context.trace_buffer_size) ? 
                          g_trace_index : 0;
    
    for (uint32_t i = 0; i < g_debug_context.trace_buffer_size; i++) {
        uint32_t index = (start_index + i) % g_debug_context.trace_buffer_size;
        trace_entry_t *entry = &g_trace_buffer[index];
        
        if (entry->timestamp == 0) continue;  // Empty entry
        
        const char *level_str[] = {"ERROR", "WARNING", "INFO", "DEBUG", "TRACE"};
        DEBUG_INFO("[%u] %s %s:%u %s() - %s", 
                   entry->timestamp, level_str[entry->level],
                   entry->file, entry->line, entry->function, entry->message);
    }
    
    DEBUG_INFO("=== End Trace Dump ===");
}
