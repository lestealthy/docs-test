Yocto Project Components

BitBake Recipe Development

BitBake recipes define how to build software packages. They contain metadata describing the source code, dependencies, and build instructions.

Basic Recipe Example 
# Example: myapp_1.0.bb
SUMMARY = "Custom Application for Embedded System"
DESCRIPTION = "A sample application demonstrating Yocto Project recipe development"
LICENSE = "MIT"
LIC_FILES_CHKSUM = "file://LICENSE;md5=1234567890abcdef"

# Source code location
SRC_URI = "git://github.com/user/myapp.git;protocol=https;branch=main"
SRCREV = "a1b2c3d4e5f6789012345678901234567890abcd"

# Build dependencies
DEPENDS = "glib-2.0 json-c"

# Runtime dependencies  
RDEPENDS:${PN} = "glib-2.0 json-c"

# Build configuration
inherit pkgconfig autotools

# Install additional files
do_install:append() {
    install -d ${D}${sysconfdir}/myapp
    install -m 0644 ${S}/config/myapp.conf ${D}${sysconfdir}/myapp/
    
    install -d ${D}${systemd_system_unitdir}
    install -m 0644 ${S}/systemd/myapp.service ${D}${systemd_system_unitdir}/
}

# Enable systemd service
SYSTEMD_SERVICE:${PN} = "myapp.service"

$ bitbake-layers create-layer meta-mycompany

├── conf/
│   ├── layer.conf           # Layer configuration
│   └── bblayers.conf        # Layer dependencies

│       └── myapp_1.0.bb     # Application recipe
├── recipes-kernel/
│   └── linux/
│       └── linux-yocto_%.bbappend  # Kernel modifications

    char payload[256];
    struct json_obj_descr sensor_descr[] = {
        JSON_OBJ_DESCR_PRIM(struct sensor_data, timestamp, JSON_TOK_NUMBER),
        JSON_OBJ_DESCR_PRIM(struct sensor_data, temperature, JSON_TOK_NUMBER),
        JSON_OBJ_DESCR_PRIM(struct sensor_data, humidity, JSON_TOK_NUMBER),

    k_work_reschedule(&gw->publish_work, K_SECONDS(30));
}

    coap_send(session, pdu);
    
    coap_io_process(ctx, COAP_IO_WAIT);
    
    coap_free_context(ctx);
    return 0;
}

struct mqtt_client_context {
    struct mqtt_client client;
    struct sockaddr_storage broker;
    int sock;
    bool connected;
};

static void mqtt_evt_handler(struct mqtt_client *client,
                           const struct mqtt_evt *evt) {
    switch (evt->type) {
        case MQTT_EVT_CONNACK:
            if (evt->result != 0) {
                printk("MQTT connection failed: %d\n", evt->result);
                return;
            }
            printk("MQTT connected\n");
            mqtt_subscribe(client, "device/+/sensor", MQTT_QOS_1_AT_LEAST_ONCE);
            break;
            
        case MQTT_EVT_PUBLISH: {
            const struct mqtt_publish_param *p = &evt->param.publish;
            printk("Received: %.*s\n", p->payload.len, p->payload.data);
            mqtt_publish_qos1_ack(client, p->message_id);
            break;
        }
        
        case MQTT_EVT_DISCONNECT:
            printk("MQTT disconnected\n");
            break;
            
        default:
            break;
    }
}

static int mqtt_connect(struct mqtt_client_context *ctx) {
    struct sockaddr_in *broker4 = (struct sockaddr_in *)&ctx->broker;
    broker4->sin_family = AF_INET;
    broker4->sin_port = htons(1883);
    inet_pton(AF_INET, "192.168.1.100", &broker4->sin_addr);
    
    ctx->sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (ctx->sock < 0) return -1;
    
    if (connect(ctx->sock, (struct sockaddr *)&ctx->broker,
                sizeof(ctx->broker)) < 0) {
        close(ctx->sock);
        return -1;
    }
    
    mqtt_client_init(&ctx->client, ctx->sock, NULL, NULL, mqtt_evt_handler);
    
    struct mqtt_utf8 client_id = {.utf8 = "zephyr_client", .size = 12};
    struct mqtt_utf8 username = {.utf8 = "user", .size = 4};
    struct mqtt_utf8 password = {.utf8 = "pass", .size = 4};
    
    return mqtt_connect(&ctx->client, &client_id, &username, &password,
                       0, 0, 0, MQTT_CONNECTION_TYPE_CLEAN);
}

static int http_post_sensor_data(const char *url, const char *data) {
    struct http_request req = {0};

    addr.sin_port = htons(80);

        return -1;
    }
    
    req.method = HTTP_POST;
    req.url = url;
    req.host = "192.168.1.100";
    req.protocol = "HTTP/1.1";
    req.payload = data;
    req.payload_len = strlen(data);
    req.header_fields = "Content-Type: application/json\r\n";
    
    int ret = http_client_request(sock, &req, 5000, &rsp);
    close(sock);
    
    return ret;
}

static void websocket_evt_handler(struct websocket_context *ctx,

    switch (type) {
        case WEBSOCKET_EVT_CONNECTED:
            printk("WebSocket connected\n");
            ctx->connected = true;
            break;
            
        case WEBSOCKET_EVT_MSG_RECEIVED:
            printk("Received: %.*s\n", evt->msg.len, evt->msg.data);
            break;
            
        case WEBSOCKET_EVT_DISCONNECTED:
            printk("WebSocket disconnected\n");
            ctx->connected = false;
            break;
            
        default:
            break;
    }
}

static int websocket_connect(struct websocket_context *ctx) {
    struct sockaddr_in addr;
    
    ctx->sock = socket(AF_INET, SOCK_STREAM, IPPROTO_TCP);
    if (ctx->sock < 0) return -1;
    
    addr.sin_family = AF_INET;
    addr.sin_port = htons(8080);
    inet_pton(AF_INET, "192.168.1.100", &addr.sin_addr);

        close(ctx->sock);
        return -1;
    }
    
    ctx->req.host = "192.168.1.100:8080";
    ctx->req.url = "/ws";
    ctx->req.protocol = "websocket";

Data Aggregation

Real-time data processing and analytics:

Stream Processing 
typedef struct {
    float temperature;
    float humidity;
    uint64_t timestamp;
    uint32_t count;
} sensor_window_t;

typedef struct {
    sensor_window_t windows[10];
    uint8_t current_window;
    uint32_t window_size_ms;
    uint64_t last_aggregation;
} data_aggregator_t;

static void aggregate_sensor_data(data_aggregator_t *agg, 
                                 float temperature, float humidity) {
    uint64_t now = k_uptime_get();
    uint8_t window_index = (now / agg->window_size_ms) % 10;
    
    if (window_index != agg->current_window) {
        // New window - process previous window
        process_window(&agg->windows[agg->current_window]);
        agg->current_window = window_index;
        agg->windows[window_index].count = 0;
    }
    
    sensor_window_t *window = &agg->windows[window_index];
    window->temperature += temperature;
    window->humidity += humidity;
    window->timestamp = now;
    window->count++;
}

static void process_window(sensor_window_t *window) {
    if (window->count == 0) return;
    
    float avg_temp = window->temperature / window->count;
    float avg_humidity = window->humidity / window->count;
    
    // Send aggregated data to cloud
    char json_data[256];
    snprintf(json_data, sizeof(json_data),
             "{\"timestamp\":%lu,\"avg_temperature\":%.2f,\"avg_humidity\":%.2f,\"count\":%u}",
             window->timestamp, avg_temp, avg_humidity, window->count);
    
    http_post_sensor_data("/api/aggregated", json_data);
}

int ml_model_init(const uint8_t *model_data, size_t model_size) {
    const tflite::Model *model = tflite::GetModel(model_data);
    if (model->version() != TFLITE_SCHEMA_VERSION) {
        return -1;
    }
    
    static tflite::MicroInterpreter static_interpreter(
        model, resolver, tensor_arena, kTensorArenaSize, µ_error_reporter);
    interpreter = &static_interpreter;
    
    TfLiteStatus allocate_status = interpreter->AllocateTensors();
    if (allocate_status != kTfLiteOk) {
        return -1;
    }
    
    input = interpreter->input(0);
    output = interpreter->output(0);
    
    return 0;
}

float ml_predict(float *input_data, size_t input_size) {
    if (input->bytes != input_size * sizeof(float)) {
        return -1.0f;
    }
    
    memcpy(input->data.f, input_data, input->bytes);
    
    TfLiteStatus invoke_status = interpreter->Invoke();
    if (invoke_status != kTfLiteOk) {
        return -1.0f;
    }
    
    return output->data.f[0];
}

typedef struct {
    char device_id[32];
    uint8_t private_key[32];
    char public_key[64];
    char token[512];
    uint64_t token_expiry;
} device_auth_t;

static int generate_jwt_token(device_auth_t *auth) {
    uint64_t now = k_uptime_get();
    uint64_t expiry = now + (24 * 60 * 60 * 1000); // 24 hours
    
    char header[] = "{\"alg\":\"ES256\",\"typ\":\"JWT\"}";
    char payload[256];
    snprintf(payload, sizeof(payload),
             "{\"iss\":\"%s\",\"exp\":%llu,\"iat\":%llu}",
             auth->device_id, expiry, now);
    
    // Base64 encode header and payload
    char encoded_header[128];
    char encoded_payload[512];
    base64_encode(header, strlen(header), encoded_header);
    base64_encode(payload, strlen(payload), encoded_payload);
    
    // Create signature
    char message[1024];
    snprintf(message, sizeof(message), "%s.%s", encoded_header, encoded_payload);
    
    uint8_t signature[64];
    int ret = crypto_sign_ed25519(auth->private_key, (uint8_t*)message,
                                 strlen(message), signature);
    if (ret != 0) return -1;
    
    char encoded_signature[128];
    base64_encode(signature, 64, encoded_signature);
    
    snprintf(auth->token, sizeof(auth->token), "%s.%s.%s",
             encoded_header, encoded_payload, encoded_signature);
    
    auth->token_expiry = expiry;
    return 0;
}

static bool is_token_valid(device_auth_t *auth) {
    return k_uptime_get() < auth->token_expiry;
}

typedef struct {
    uint8_t key[32];
    uint8_t iv[16];
    struct cipher_ctx ctx;
} encryption_context_t;

static int encrypt_sensor_data(encryption_context_t *ctx,
                               const uint8_t *plaintext, size_t plaintext_len,
                               uint8_t *ciphertext, size_t *ciphertext_len) {
    // Generate random IV
    sys_csrand_get(ctx->iv, sizeof(ctx->iv));
    
    // Initialize cipher context
    ctx->ctx.keylen = 32;
    ctx->ctx.key.bit_stream = ctx->key;
    
    int ret = cipher_begin_session(ctx->ctx.device, &ctx->ctx,
                                  CRYPTO_CIPHER_ALGO_AES,
                                  CRYPTO_CIPHER_MODE_CBC,
                                  CRYPTO_CIPHER_OP_ENCRYPT);
    if (ret != 0) return ret;
    
    // Encrypt data
    ret = cipher_cbc_op(&ctx->ctx, ctx->iv, plaintext, plaintext_len,
                        ciphertext, ciphertext_len);
    
    cipher_free_session(ctx->ctx.device, &ctx->ctx);
    return ret;
}

static int decrypt_sensor_data(encryption_context_t *ctx,
                               const uint8_t *ciphertext, size_t ciphertext_len,
                               uint8_t *plaintext, size_t *plaintext_len) {
    ctx->ctx.keylen = 32;
    ctx->ctx.key.bit_stream = ctx->key;
    
    int ret = cipher_begin_session(ctx->ctx.device, &ctx->ctx,
                                  CRYPTO_CIPHER_ALGO_AES,
                                  CRYPTO_CIPHER_MODE_CBC,
                                  CRYPTO_CIPHER_OP_DECRYPT);
    if (ret != 0) return ret;
    
    ret = cipher_cbc_op(&ctx->ctx, ctx->iv, ciphertext, ciphertext_len,
                        plaintext, plaintext_len);
    
    cipher_free_session(ctx->ctx.device, &ctx->ctx);
    return ret;
}

Secure Boot

Verifying firmware integrity:

Secure Boot Implementation 
#include 

#define BOOT_MAGIC 0xDEADBEEF
#define FIRMWARE_SIGNATURE_SIZE 256

typedef struct {
    uint32_t magic;
    uint32_t version;
    uint32_t size;
    uint8_t signature[FIRMWARE_SIGNATURE_SIZE];
    uint8_t hash[32];
} firmware_header_t;

static bool verify_firmware_signature(const firmware_header_t *header,
                                     const uint8_t *firmware_data) {
    // Verify magic number
    if (header->magic != BOOT_MAGIC) {
        return false;
    }
    
    // Calculate SHA-256 hash of firmware
    uint8_t calculated_hash[32];
    struct hash_ctx ctx;
    
    int ret = hash_begin_session(crypto_dev, &ctx, CRYPTO_HASH_ALGO_SHA256);
    if (ret != 0) return false;
    
    ret = hash_update(&ctx, firmware_data, header->size);
    if (ret != 0) {
        hash_free_session(crypto_dev, &ctx);
        return false;
    }
    
    ret = hash_final(&ctx, calculated_hash);
    hash_free_session(crypto_dev, &ctx);
    
    if (ret != 0) return false;
    
    // Compare hashes
    if (memcmp(header->hash, calculated_hash, 32) != 0) {
        return false;
    }
    
    // Verify digital signature
    ret = crypto_verify_ed25519(public_key, header->signature,
                               FIRMWARE_SIGNATURE_SIZE, firmware_data,
                               header->size);
    
    return ret == 0;
}

static bool secure_boot_verify(void) {
    const firmware_header_t *header = (const firmware_header_t*)0x08000000;
    const uint8_t *firmware = (const uint8_t*)(0x08000000 + sizeof(firmware_header_t));
    
    if (!verify_firmware_signature(header, firmware)) {
        // Boot into recovery mode
        jump_to_recovery();
        return false;
    }
    
    return true;
}
