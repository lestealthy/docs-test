# Custom commands
proc read_memory { addr len } {
    set data ""
    for {set i 0} {$i < $len} {incr i 4} {
        set val [mdw [expr $addr + $i]]
        append data [format "%08x " $val]
    }
    return $data
}

Technical Standards

Industry standards and specifications:

MISRA C Guidelines 
// MISRA C:2012 Guidelines for Embedded Systems
// Rule 8.2: All identifiers shall be unique
// Rule 8.3: All declarations shall be at file scope or block scope
// Rule 8.4: Compatible types shall be used for parameters and return values

// Rule 8.6: Identifiers shall be distinct from macro names
#define MAX_SIZE 100
// int MAX_SIZE; // Violation: identifier same as macro

// Rule 9.1: The value of an object with automatic storage duration shall not be read before it has been set
void function(void) {
    int x; // Uninitialized
    // int y = x; // Violation: reading uninitialized variable
    x = 5;  // Correct: initialize before use
    int y = x; // Correct: x is now initialized
}

// Rule 10.1: Operands shall not be of an inappropriate essential type
uint8_t a = 5;
uint8_t b = 10;
// uint8_t c = a + b; // Violation: result may overflow
uint16_t c = (uint16_t)a + (uint16_t)b; // Correct: explicit cast

// Rule 10.3: The value of an expression shall not be assigned to an object with a narrower essential type
uint16_t value = 0x1234;
uint8_t byte = (uint8_t)value; // Correct: explicit cast

// Rule 11.1: Conversions shall not be performed between a pointer to function and any other type
void (*func_ptr)(void) = NULL;
// int *int_ptr = (int*)func_ptr; // Violation: inappropriate conversion

// Rule 11.2: Conversions shall not be performed between a pointer to object and any other type
int *int_ptr = NULL;
// void (*func_ptr)(void) = (void(*)())int_ptr; // Violation

// Rule 11.3: A cast shall not be performed between a pointer to object type and a pointer to a different object type
int *int_ptr = NULL;
// char *char_ptr = (char*)int_ptr; // Violation: different object types
// Use union for type punning if necessary

// Rule 12.1: The precedence of operators within expressions shall be made explicit
int result = a + b * c; // Violation: precedence not explicit
int result = a + (b * c); // Correct: explicit precedence

// Rule 13.1: Initializer lists shall not contain persistent side effects
int array[] = {1, 2, 3, 4, 5}; // Correct: no side effects
// int array[] = {1, 2, function(), 4, 5}; // Violation: side effect in initializer

// Rule 14.1: A loop counter shall not have essentially floating type
// for (float i = 0.0f; i < 10.0f; i += 1.0f) // Violation: floating counter
for (int i = 0; i < 10; i++) // Correct: integer counter

// Rule 15.1: The goto statement shall not be used
// goto label; // Violation: avoid goto
// Use structured programming constructs instead

// Rule 16.1: All switch statements shall be well-formed
switch (value) {
    case 1:
        // statements
        break;
    case 2:
        // statements
        break;
    default:
        // statements
        break;
}

// Rule 17.1: The features of  shall not be used
// va_list args; // Violation: avoid variadic functions
// Use function overloading or fixed parameter lists

// Rule 18.1: A pointer resulting from arithmetic on a pointer operand shall address an element of the same array
int array[10];
int *ptr = array;
// int *end = ptr + 20; // Violation: beyond array bounds
int *end = ptr + 10; // Correct: within array bounds

/**
 * @file sensor_api.h
 * @brief Sensor API for embedded systems
 * @author WattLab Team
 * @version 1.0.0
 * @date 2024-01-01
 */

#ifndef SENSOR_API_H
#define SENSOR_API_H

#include 
#include 

/**
 * @defgroup sensor_api Sensor API
 * @brief Comprehensive sensor interface for embedded systems
 * @{
 */

/**
 * @brief Sensor types supported by the API
 */
typedef enum {
    SENSOR_TYPE_TEMPERATURE,    /**< Temperature sensor */
    SENSOR_TYPE_HUMIDITY,       /**< Humidity sensor */
    SENSOR_TYPE_PRESSURE,       /**< Pressure sensor */
    SENSOR_TYPE_ACCELEROMETER, /**< Accelerometer sensor */
    SENSOR_TYPE_GYROSCOPE,      /**< Gyroscope sensor */
    SENSOR_TYPE_MAGNETOMETER    /**< Magnetometer sensor */
} sensor_type_t;

/**
 * @brief Sensor configuration structure
 */
typedef struct {
    sensor_type_t type;         /**< Sensor type */
    uint32_t sample_rate;       /**< Sample rate in Hz */
    uint8_t resolution;         /**< Resolution in bits */
    bool enable_interrupt;      /**< Enable interrupt on data ready */
    void (*callback)(void);     /**< Data ready callback function */
} sensor_config_t;

/**
 * @brief Sensor data structure
 */
typedef struct {
    float value;                /**< Sensor value */
    uint64_t timestamp;        /**< Timestamp in microseconds */
    uint8_t quality;           /**< Data quality (0-100) */
} sensor_data_t;

/**
 * @brief Initialize sensor with given configuration
 * @param config Pointer to sensor configuration
 * @return 0 on success, negative error code on failure
 * @retval 0 Success
 * @retval -EINVAL Invalid configuration
 * @retval -ENODEV Sensor not found
 * @retval -EBUSY Sensor already initialized
 * 
 * @code
 * sensor_config_t config = {
 *     .type = SENSOR_TYPE_TEMPERATURE,
 *     .sample_rate = 10,
 *     .resolution = 12,
 *     .enable_interrupt = true,
 *     .callback = temperature_callback
 * };
 * 
 * int ret = sensor_init(&config);
 * if (ret < 0) {
 *     printf("Sensor initialization failed: %d\n", ret);
 * }
 * @endcode
 */
int sensor_init(const sensor_config_t *config);

/**
 * @brief Read sensor data
 * @param data Pointer to store sensor data
 * @return 0 on success, negative error code on failure
 * @retval 0 Success
 * @retval -ENODEV Sensor not initialized
 * @retval -EAGAIN No data available
 * 
 * @code
 * sensor_data_t data;
 * int ret = sensor_read(&data);
 * if (ret == 0) {
 *     printf("Temperature: %.2f°C\n", data.value);
 * }
 * @endcode
 */
int sensor_read(sensor_data_t *data);

/**
 * @brief Set sensor sample rate
 * @param rate Sample rate in Hz
 * @return 0 on success, negative error code on failure
 */
int sensor_set_sample_rate(uint32_t rate);

# Embedded Systems Professional Certifications

## ARM Certifications
### ARM Accredited Engineer (AAE)
- Prerequisites: 2+ years embedded experience
- Exam: ARM Cortex-M programming and architecture
- Topics: Assembly, C programming, debugging, optimization
- Validity: 3 years
- Cost: $200

### ARM Certified Professional (ACP)
- Prerequisites: AAE certification
- Exam: Advanced ARM system design
- Topics: Multi-core systems, security, performance optimization
- Validity: 3 years
- Cost: $300

## IEEE Certifications
### Certified Software Development Professional (CSDP)
- Prerequisites: Bachelor's degree + 2 years experience
- Exam: Software engineering principles
- Topics: Requirements, design, testing, maintenance
- Validity: 3 years
- Cost: $400

### Certified Software Quality Engineer (CSQE)
- Prerequisites: Bachelor's degree + 4 years experience
- Exam: Quality assurance and testing
- Topics: Test planning, execution, metrics, tools
- Validity: 3 years
- Cost: $400

## CompTIA Certifications
### CompTIA Security+
- Prerequisites: 2 years IT security experience
- Exam: Security fundamentals
- Topics: Threats, vulnerabilities, cryptography, access control
- Validity: 3 years
- Cost: $370

### CompTIA Network+
- Prerequisites: 9 months networking experience
- Exam: Network technologies
- Topics: Protocols, troubleshooting, security, wireless
- Validity: 3 years
- Cost: $338

## Vendor-Specific Certifications
### STMicroelectronics STM32 Certification
- Prerequisites: Basic embedded programming
- Exam: STM32 ecosystem and tools
- Topics: HAL, CubeMX, debugging, optimization
- Validity: 2 years
- Cost: $150

### Espressif ESP32 Certification
- Prerequisites: IoT development experience
- Exam: ESP32 programming and IoT
- Topics: WiFi, Bluetooth, FreeRTOS, security
- Validity: 2 years
- Cost: $100

### NXP i.MX Certification
- Prerequisites: Linux embedded experience
- Exam: i.MX processor programming
- Topics: Linux kernel, device drivers, Yocto
- Validity: 2 years
- Cost: $200
